-- LOADLINK PROFESSIONAL REPAIR — FOLLOWED PROFILES AND NOTIFICATION PREFERENCES
-- Run this complete file in a NEW Supabase SQL Editor query.
-- Safe to run more than once. It does not delete users, listings, chats or existing follows.

begin;

create table if not exists public.loadlink_profile_follows (
  user_id uuid not null references auth.users(id) on delete cascade,
  target_type text not null check (target_type in ('dealership','contractor','opportunity_poster','user')),
  target_id text not null,
  target_name text not null,
  target_href text not null,
  target_location text,
  target_image text,
  notify_new_listings boolean not null default true,
  notify_updates boolean not null default true,
  notify_price_changes boolean not null default false,
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now(),
  primary key (user_id, target_type, target_id)
);

alter table public.loadlink_profile_follows
  add column if not exists target_location text,
  add column if not exists target_image text,
  add column if not exists notify_new_listings boolean not null default true,
  add column if not exists notify_updates boolean not null default true,
  add column if not exists notify_price_changes boolean not null default false,
  add column if not exists created_at timestamptz not null default now(),
  add column if not exists updated_at timestamptz not null default now();

create index if not exists loadlink_profile_follows_user_updated_idx
  on public.loadlink_profile_follows(user_id, updated_at desc);

alter table public.loadlink_profile_follows enable row level security;

revoke all on public.loadlink_profile_follows from anon;
grant select, insert, update, delete on public.loadlink_profile_follows to authenticated;

drop policy if exists "loadlink users read own follows" on public.loadlink_profile_follows;
create policy "loadlink users read own follows"
on public.loadlink_profile_follows
for select to authenticated
using (user_id = auth.uid());

drop policy if exists "loadlink users create own follows" on public.loadlink_profile_follows;
create policy "loadlink users create own follows"
on public.loadlink_profile_follows
for insert to authenticated
with check (user_id = auth.uid());

drop policy if exists "loadlink users update own follows" on public.loadlink_profile_follows;
create policy "loadlink users update own follows"
on public.loadlink_profile_follows
for update to authenticated
using (user_id = auth.uid())
with check (user_id = auth.uid());

drop policy if exists "loadlink users delete own follows" on public.loadlink_profile_follows;
create policy "loadlink users delete own follows"
on public.loadlink_profile_follows
for delete to authenticated
using (user_id = auth.uid());

create or replace function public.loadlink_touch_profile_follow()
returns trigger
language plpgsql
set search_path = public, pg_temp
as $$
begin
  new.updated_at := now();
  return new;
end;
$$;

drop trigger if exists loadlink_profile_follows_touch on public.loadlink_profile_follows;
create trigger loadlink_profile_follows_touch
before update on public.loadlink_profile_follows
for each row execute function public.loadlink_touch_profile_follow();

commit;
