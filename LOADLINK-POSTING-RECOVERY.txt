-- LOADLINK POSTING RECOVERY
-- One safe, repeatable repair for job, contract, asset and paid vehicle posting.
-- It does not delete listings, users, messages, payments or uploaded media.
-- Run once in Supabase SQL Editor after deploying the matching website update.

begin;

create extension if not exists pgcrypto;

grant usage on schema public to anon, authenticated;

create table if not exists public.job_listings (
  id uuid primary key default gen_random_uuid(),
  title text not null,
  city text not null,
  vehicle_group text not null,
  rate text not null,
  posted_by text not null,
  contact_number text not null default '',
  description text not null,
  photos text[] not null default '{}'::text[],
  sponsored boolean not null default false,
  package_type text not null default 'standard',
  created_at timestamptz not null default now()
);

alter table public.job_listings
  add column if not exists whatsapp_number text not null default '',
  add column if not exists poster_photo text not null default '',
  add column if not exists owner_key text not null default '',
  add column if not exists user_id uuid references auth.users(id) on delete set null,
  add column if not exists listing_kind text not null default 'job',
  add column if not exists status text not null default 'active',
  add column if not exists moderation_status text not null default 'pending',
  add column if not exists moderation_notes text,
  add column if not exists moderated_at timestamptz,
  add column if not exists moderated_by uuid references auth.users(id) on delete set null,
  add column if not exists expires_at timestamptz,
  add column if not exists payment_status text not null default 'not_required',
  add column if not exists client_request_id text;

-- Keep blocked/suspended accounts blocked even though the discontinued NDA gate
-- is removed from listing submission.
create table if not exists public.user_moderation_profiles (
  user_id uuid primary key references auth.users(id) on delete cascade,
  status text not null default 'active' check (status in ('active','flagged','suspended','blocked')),
  reason text,
  internal_notes text,
  strike_count integer not null default 0 check (strike_count >= 0),
  suspended_until timestamptz,
  last_action_by uuid references auth.users(id) on delete set null,
  last_action_at timestamptz not null default now(),
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now()
);

alter table public.user_moderation_profiles
  add column if not exists status text not null default 'active',
  add column if not exists suspended_until timestamptz;

create or replace function public.loadlink_account_status(p_user_id uuid default auth.uid())
returns text
language sql
stable
security definer
set search_path = public, pg_temp
as $$
  select case
    when p_user_id is null then 'active'
    when coalesce((select status from public.user_moderation_profiles where user_id=p_user_id),'active')='suspended'
      and coalesce((select suspended_until from public.user_moderation_profiles where user_id=p_user_id), now()+interval '1 day') <= now()
      then 'active'
    else coalesce((select status from public.user_moderation_profiles where user_id=p_user_id),'active')
  end;
$$;

revoke all on function public.loadlink_account_status(uuid) from public;
grant execute on function public.loadlink_account_status(uuid) to authenticated;

-- The NDA was scrapped. Remove only obsolete listing triggers connected to the
-- old NDA/legacy posting guards. Current ownership, payment and moderation
-- triggers remain active.
do $$
declare
  trigger_row record;
begin
  for trigger_row in
    select t.tgname
    from pg_trigger t
    join pg_class c on c.oid = t.tgrelid
    join pg_namespace n on n.oid = c.relnamespace
    join pg_proc p on p.oid = t.tgfoid
    where not t.tgisinternal
      and n.nspname = 'public'
      and c.relname = 'job_listings'
      and p.proname in (
        'loadlink_enforce_listing_rules',
        'loadlink_enforce_platform_access_trigger'
      )
  loop
    execute format('drop trigger if exists %I on public.job_listings', trigger_row.tgname);
  end loop;
end
$$;

alter table public.job_listings disable trigger user;

alter table public.job_listings
  drop constraint if exists job_listings_status_check;

alter table public.job_listings
  drop constraint if exists job_listings_moderation_status_check;

update public.job_listings
set status = case
  when status is null or btrim(status) = '' then 'active'
  when lower(btrim(status)) in ('active','published','approved','pending','pending_review','live','available') then 'active'
  when lower(btrim(status)) = 'draft' then 'draft'
  when lower(btrim(status)) in ('filled','complete','completed') then 'filled'
  when lower(btrim(status)) in ('closed','archived','expired','rejected','removed') then 'closed'
  else 'active'
end,
moderation_status = case
  when moderation_status is null or btrim(moderation_status) = '' then 'pending'
  when lower(btrim(moderation_status)) in ('approved','accepted','published','live') then 'approved'
  when lower(btrim(moderation_status)) in ('rejected','declined','denied') then 'rejected'
  else 'pending'
end,
listing_kind = case
  when listing_kind is null or btrim(listing_kind) = '' then 'job'
  when lower(btrim(listing_kind)) in ('job','contract','asset','vehicle') then lower(btrim(listing_kind))
  when lower(btrim(listing_kind)) in ('truck_sale','vehicle_listing') then 'vehicle'
  else 'job'
end,
photos = coalesce(photos, '{}'::text[]),
package_type = coalesce(nullif(btrim(package_type), ''), 'standard'),
payment_status = coalesce(nullif(btrim(payment_status), ''), 'not_required');

alter table public.job_listings
  alter column status set default 'active',
  alter column moderation_status set default 'pending',
  alter column listing_kind set default 'job',
  alter column package_type set default 'standard',
  alter column payment_status set default 'not_required';

alter table public.job_listings
  add constraint job_listings_status_check
  check (status in ('active','filled','closed','draft'));

alter table public.job_listings
  add constraint job_listings_moderation_status_check
  check (moderation_status in ('pending','approved','rejected'));

alter table public.job_listings enable trigger user;

create index if not exists job_listings_user_created_idx
  on public.job_listings(user_id, created_at desc);

create index if not exists job_listings_request_idx
  on public.job_listings(user_id, client_request_id)
  where client_request_id is not null;

create index if not exists job_listings_public_recovery_idx
  on public.job_listings(moderation_status, status, expires_at, created_at desc);

alter table public.job_listings enable row level security;

-- Public users see approved active listings. Signed-in owners also see their
-- pending listings in My Posts and can recover a timed-out submission by its ID.
drop policy if exists "loadlink_jobs_read_all" on public.job_listings;
drop policy if exists "loadlink_jobs_read_active_or_own" on public.job_listings;
drop policy if exists "loadlink_jobs_read_approved_or_own" on public.job_listings;
drop policy if exists "loadlink_jobs_read_approved_or_own_v2" on public.job_listings;
drop policy if exists "loadlink_jobs_read_approved_or_own_recovery" on public.job_listings;
create policy "loadlink_jobs_read_approved_or_own_recovery"
on public.job_listings
for select
to anon, authenticated
using (
  (
    moderation_status = 'approved'
    and status = 'active'
    and (expires_at is null or expires_at > now())
  )
  or user_id = auth.uid()
);

-- Direct browser fallback used when PostgREST has not refreshed the RPC schema.
drop policy if exists "loadlink_jobs_insert_all_testing" on public.job_listings;
drop policy if exists "loadlink_jobs_insert_authenticated" on public.job_listings;
drop policy if exists "loadlink_jobs_insert_authenticated_v2" on public.job_listings;
drop policy if exists "loadlink_jobs_insert_authenticated_recovery" on public.job_listings;
create policy "loadlink_jobs_insert_authenticated_recovery"
on public.job_listings
for insert
to authenticated
with check (
  auth.uid() is not null
  and user_id = auth.uid()
  and status = 'active'
  and moderation_status = 'pending'
  and public.loadlink_account_status(auth.uid()) not in ('blocked','suspended')
);

grant select on public.job_listings to anon;
grant select, insert on public.job_listings to authenticated;

-- Public media reads with authenticated owner-folder writes.
insert into storage.buckets (id, name, public, file_size_limit, allowed_mime_types)
values (
  'job-photos',
  'job-photos',
  true,
  6291456,
  array['image/jpeg','image/png','image/webp']
)
on conflict (id) do update
set public = excluded.public,
    file_size_limit = excluded.file_size_limit,
    allowed_mime_types = excluded.allowed_mime_types;

drop policy if exists "loadlink_job_photos_read_all" on storage.objects;
create policy "loadlink_job_photos_read_all"
on storage.objects
for select
to public
using (bucket_id = 'job-photos');

drop policy if exists "loadlink_job_photos_insert_all_testing" on storage.objects;
drop policy if exists "loadlink_job_photos_owner_insert_v2" on storage.objects;
drop policy if exists "loadlink_job_photos_owner_insert_recovery" on storage.objects;
create policy "loadlink_job_photos_owner_insert_recovery"
on storage.objects
for insert
to authenticated
with check (
  bucket_id = 'job-photos'
  and (storage.foldername(name))[1] = auth.uid()::text
  and public.loadlink_account_status(auth.uid()) not in ('blocked','suspended')
);

drop policy if exists "loadlink_job_photos_owner_update_v2" on storage.objects;
drop policy if exists "loadlink_job_photos_owner_update_recovery" on storage.objects;
create policy "loadlink_job_photos_owner_update_recovery"
on storage.objects
for update
to authenticated
using (
  bucket_id = 'job-photos'
  and (storage.foldername(name))[1] = auth.uid()::text
)
with check (
  bucket_id = 'job-photos'
  and (storage.foldername(name))[1] = auth.uid()::text
);

drop policy if exists "loadlink_job_photos_owner_delete_v2" on storage.objects;
drop policy if exists "loadlink_job_photos_owner_delete_recovery" on storage.objects;
create policy "loadlink_job_photos_owner_delete_recovery"
on storage.objects
for delete
to authenticated
using (
  bucket_id = 'job-photos'
  and (storage.foldername(name))[1] = auth.uid()::text
);

-- Authenticated, idempotent posting RPC. The website also contains a direct
-- insert fallback, so posting survives a temporary PostgREST schema-cache delay.
drop function if exists public.loadlink_submit_listing_v2(
  text,text,text,text,text,text,text,text,text,text[],text,text,text
);

create function public.loadlink_submit_listing_v2(
  p_title text,
  p_city text,
  p_vehicle_group text,
  p_rate text,
  p_posted_by text,
  p_contact_number text,
  p_whatsapp_number text,
  p_poster_photo text,
  p_description text,
  p_photos text[],
  p_listing_kind text,
  p_client_request_id text,
  p_owner_key text
)
returns uuid
language plpgsql
security definer
set search_path = public, pg_temp
as $$
declare
  v_user uuid := auth.uid();
  v_listing_id uuid;
  v_kind text;
  v_contact text;
  v_whatsapp text;
  v_photo_count integer := coalesce(array_length(p_photos,1),0);
begin
  if v_user is null then
    raise exception using errcode='28000', message='Authentication required';
  end if;

  if public.loadlink_account_status(v_user) in ('blocked','suspended') then
    raise exception using errcode='42501', message='Account access is restricted';
  end if;

  if char_length(btrim(coalesce(p_title,''))) < 2 then
    raise exception using errcode='22023', message='Enter a valid listing title';
  end if;
  if char_length(btrim(coalesce(p_city,''))) < 2 then
    raise exception using errcode='22023', message='Enter a valid listing location';
  end if;
  if char_length(btrim(coalesce(p_vehicle_group,''))) < 2 then
    raise exception using errcode='22023', message='Choose a valid vehicle group';
  end if;
  if char_length(btrim(coalesce(p_rate,''))) < 1 then
    raise exception using errcode='22023', message='Enter a rate or budget';
  end if;
  if char_length(btrim(coalesce(p_posted_by,''))) < 2 then
    raise exception using errcode='22023', message='Enter a name or company';
  end if;
  if char_length(btrim(coalesce(p_description,''))) < 2 then
    raise exception using errcode='22023', message='Enter a listing description';
  end if;

  v_contact := regexp_replace(coalesce(p_contact_number,''), '[^0-9+]', '', 'g');
  if not (v_contact ~ '^0[0-9]{9}$' or v_contact ~ '^\+27[0-9]{9}$') then
    raise exception using errcode='22023', message='Enter a valid South African contact number';
  end if;

  v_whatsapp := regexp_replace(coalesce(p_whatsapp_number,''), '[^0-9+]', '', 'g');
  if v_whatsapp <> '' and not (v_whatsapp ~ '^0[0-9]{9}$' or v_whatsapp ~ '^\+27[0-9]{9}$') then
    raise exception using errcode='22023', message='Enter a valid WhatsApp number';
  end if;

  if v_photo_count < 1 or v_photo_count > 5 then
    raise exception using errcode='22023', message='Job, contract and asset listings require 1 to 5 photos';
  end if;

  v_kind := case lower(btrim(coalesce(p_listing_kind,'job')))
    when 'contract' then 'contract'
    when 'asset' then 'asset'
    else 'job'
  end;

  if char_length(btrim(coalesce(p_client_request_id,''))) >= 8 then
    select id into v_listing_id
    from public.job_listings
    where user_id=v_user
      and client_request_id=btrim(p_client_request_id)
    order by created_at desc
    limit 1;

    if found then return v_listing_id; end if;
  end if;

  insert into public.job_listings (
    title,city,vehicle_group,rate,posted_by,contact_number,
    whatsapp_number,poster_photo,description,photos,sponsored,
    package_type,owner_key,user_id,listing_kind,status,
    moderation_status,moderation_notes,moderated_at,moderated_by,
    payment_status,client_request_id
  ) values (
    btrim(p_title),btrim(p_city),btrim(p_vehicle_group),btrim(p_rate),
    btrim(p_posted_by),v_contact,v_whatsapp,btrim(coalesce(p_poster_photo,'')),
    btrim(p_description),p_photos,false,'standard',left(coalesce(p_owner_key,''),500),
    v_user,v_kind,'active','pending',null,null,null,'not_required',
    nullif(btrim(coalesce(p_client_request_id,'')),'')
  )
  returning id into v_listing_id;

  return v_listing_id;
end;
$$;

revoke all on function public.loadlink_submit_listing_v2(
  text,text,text,text,text,text,text,text,text,text[],text,text,text
) from public;

grant execute on function public.loadlink_submit_listing_v2(
  text,text,text,text,text,text,text,text,text,text[],text,text,text
) to authenticated;

notify pgrst, 'reload schema';

commit;
